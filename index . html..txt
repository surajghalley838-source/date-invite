    <script>
        let selections = {
            date: '',
            time: '',
            food: '',
            vibe: ''
        };

        // Screen switching flow mechanics
        function nextScreen(screenNum) {
            document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
            document.getElementById('screen' + screenNum).classList.add('active');

            if(screenNum === 6) {
                compileSummary();
            }
        }

        // Running away logic for the 'No' button
        function moveNoButton() {
            const noBtn = document.getElementById('noBtn');
            // Constraints to keep the button bouncing strictly within the card boundaries
            const x = Math.random() * 160 + 10;
            const y = Math.random() * 300 - 150; 
            
            noBtn.style.left = x + 'px';
            noBtn.style.top = y + 'px';
        }

        // Generic grid picker selector logic
        function selectItem(element, category, value) {
            const parent = element.parentElement;
            parent.querySelectorAll('.grid-item').forEach(item => item.classList.remove('selected'));
            element.classList.add('selected');
            selections[category] = value;
        }

        // Require selecting an item before moving forward
        function validateSelection(nextStep, category) {
            if (!selections[category]) {
                alert("Pick one first! 😉");
                return;
            }
            nextScreen(nextStep);
        }

        // Dynamic details construction on the final confirmation card
        function compileSummary() {
            const dateVal = new Date(document.getElementById('dateInput').value);
            const options = { month: 'long', day: 'numeric', year: 'numeric' };
            const formattedDate = dateVal.toLocaleDateString('en-US', options);
            
            const timeVal = document.getElementById('timeInput').value;

            document.getElementById('summaryDate').innerHTML = `📅 <b>Date:</b> ${formattedDate} @ ${timeVal}`;
            document.getElementById('summaryFood').innerHTML = `🍔 <b>Food:</b> ${selections.food}`;
            document.getElementById('summaryVibe').innerHTML = `🌴 <b>Activity:</b> ${selections.vibe}`;
        }
    </script>
